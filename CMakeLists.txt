cmake_minimum_required(VERSION 3.21)
project(TROLLfxVortexLab VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(vortex_audio src/audio/wav.cpp src/audio/audio_session.cpp src/dsp/morph_engine.cpp)
target_include_directories(vortex_audio PUBLIC src)
if(MSVC)
  target_compile_options(vortex_audio PRIVATE /W4 /permissive-)
else()
  target_compile_options(vortex_audio PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(vortex-render tools/audio_render/main.cpp)
target_link_libraries(vortex-render PRIVATE vortex_audio)

if(WIN32)
  add_library(vortex_winmm src/audio/winmm_output.cpp)
  target_link_libraries(vortex_winmm PUBLIC vortex_audio winmm)
  add_executable(vortex-play tools/audio_player/winmm_player.cpp)
  target_link_libraries(vortex-play PRIVATE vortex_winmm)
  add_executable(vortex-winmm-timing-probe tools/audio_render/winmm_timing_probe.cpp)
  target_link_libraries(vortex-winmm-timing-probe PRIVATE vortex_winmm)
  add_executable(TROLLfx-Vortex-Lab WIN32 tools/gui/win32_app.cpp)
  target_link_libraries(TROLLfx-Vortex-Lab PRIVATE vortex_winmm comctl32 comdlg32 user32 gdi32)
  add_executable(vortex-gui-smoke-test tests/gui_smoke_test.cpp)
  target_link_libraries(vortex-gui-smoke-test PRIVATE user32)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(vortex-audio-tests tests/audio_tests.cpp)
  target_link_libraries(vortex-audio-tests PRIVATE vortex_audio)
  add_test(NAME vortex-audio-tests COMMAND vortex-audio-tests)
  add_executable(vortex-session-tests tests/audio_session_tests.cpp)
  target_link_libraries(vortex-session-tests PRIVATE vortex_audio)
  add_test(NAME vortex-session-tests COMMAND vortex-session-tests)
  if(WIN32)
    add_test(NAME vortex-gui-startup COMMAND vortex-gui-smoke-test $<TARGET_FILE:TROLLfx-Vortex-Lab>)
    set_tests_properties(vortex-gui-startup PROPERTIES TIMEOUT 20)
  endif()
endif()
